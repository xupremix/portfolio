\section{Progetti Selezionati}

\cvproject
  {incin: Framework di Deep Learning Type-Safe}
  {Rust, CUDA, Const Generics}
  {Framework backend-agnostic per reti neurali con verifica a tempo di compilazione.}
  {
    \begin{itemize}
      \item Progettazione di un motore deep learning ad alte prestazioni in Rust con tipi affini e const generics per garantire compatibilità di shape e rango a tempo di compilazione con zero overhead a runtime.
      \item Implementazione di grafo di differenziazione automatica, operatori Conv2d con verifica di parità dei gradienti CUDA, checkpointing e trasformazioni modulari.
    \end{itemize}
  }

\cvproject
  {ASA Autobots: Sistema Multi-Agente Ibrido (LLM + Simbolico)}
  {Node.js, PDDL, BDI, Integrazione LLM Locale}
  {Piattaforma cooperativa per compiti logistici che integra ragionamento LLM e pianificazione simbolica.}
  {
    \begin{itemize}
      \item Architettura basata sulla collaborazione real-time tra agenti autonomi BDI, integrazione di LLM locali (Llama 3.3 70B) e pianificatore simbolico PDDL per compiti logistici.
      \item Integrazione di chiamate strutturate JSON-RPC / tool use, gestione resiliente degli errori e test coverage del \textasciitilde{}91.5\% su un'architettura a 3 livelli.
    \end{itemize}
  }

\cvproject
  {signal\_image\_video: Tracciamento 3D Multi-Oggetto con Sensor Fusion}
  {Python, PyTorch, LiDAR, EKF, KITTI}
  {Sistema di tracciamento 3D con fusione camera-LiDAR valutato sul benchmark KITTI.}
  {
    \begin{itemize}
      \item Sviluppo di una pipeline di fusione sensoriale 3D LiDAR-camera combinando filtro di Kalman CTRV-EKF e algoritmo Ungherese per l'assegnazione bi-partita.
      \item Valutazione sul benchmark di guida autonoma KITTI con netta riduzione dell'RMSE rispetto alla baseline lineare.
    \end{itemize}
  }

\cvproject
  {NLU: Intent Classification \& Slot Filling Congiunto}
  {Python, PyTorch, Transformers, BERT}
  {Pipeline end-to-end di Natural Language Understanding sul dataset ATIS.}
  {
    \begin{itemize}
      \item Pipeline di training costruita da zero, fine-tuning di modelli transformer (BERT) con teste congiunte di intent classification e sequence labeling.
      \item Raggiunta un'accuratezza dell'intent pari a 0.9765 ed elevato F1 score per il rilevamento degli slot.
    \end{itemize}
  }

\cvproject
  {evolcpp: Motore di Neuroevoluzione in C++20}
  {C++20, Template Metaprogramming, CMake}
  {Porting ad alte prestazioni delle astrazioni neurali con type safety a tempo di compilazione.}
  {
    \begin{itemize}
      \item Implementazione delle astrazioni tensoriali con template metaprogramming in C++20 per garantire shape safety a tempo di compilazione senza overhead di memoria dinamica.
    \end{itemize}
  }

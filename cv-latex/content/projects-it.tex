\section{Progetti Selezionati}

\cvproject
  {incin: Sicurezza del codice ML a tempo di compilazione}
  {Rust, Machine Learning}
  {Framework di machine learning sviluppato come progetto di tesi triennale.}
  {
    \begin{itemize}
      \item Esplora controlli a tempo di compilazione per shape e invarianti dei tensori, con l'obiettivo di rendere il codice dei modelli più sicuro e comprensibile.
      \item L'obiettivo a lungo termine è aiutare agenti AI per la ricerca a produrre codice ML valido attraverso regole architetturali chiare e feedback del compilatore.
    \end{itemize}
  }

\cvproject
  {ASA Autobots: Sistema di Consegne Multi-Agente}
  {Node.js, BDI, PDDL, Integrazione LLM}
  {Progetto di gruppo che combina coordinamento tra agenti, modelli linguistici e pianificazione simbolica.}
  {
    \begin{itemize}
      \item Ho co-guidato un gruppo di due persone nello sviluppo di un flusso di consegne con agenti BDI, un coordinatore LLM e un pianificatore PDDL.
      \item Mi sono occupato dell'architettura del sistema e dell'integrazione dei componenti ad agenti e delle interfacce di servizio.
    \end{itemize}
  }

\cvproject
  {MOT Benchmark}
  {Python, PyTorch, Multi-Object Tracking, PPO}
  {Confronto empirico tra metodi classici di tracking-by-detection e un approccio multi-agente basato su PPO.}
  {
    \begin{itemize}
      \item Benchmark di SORT, OC-SORT, ByteTrack e MARLMOT in attività di tracciamento multi-oggetto.
    \end{itemize}
  }

\cvproject
  {NLU: Classificazione degli Intent e Slot Filling}
  {Python, PyTorch, BERT, Hugging Face, ATIS}
  {Progetto di comprensione del linguaggio naturale con un modello congiunto per intent e slot filling.}
  {
    \begin{itemize}

    \end{itemize}
  }
